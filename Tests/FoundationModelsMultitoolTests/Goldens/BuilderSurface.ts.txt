// tools.getWeather
/**
 * Current weather for a city.
 * @param args.city — IATA city code or city name. (required)
 * @param args.units — temperature unit; one of "c" | "f". (optional)
 * @returns Promise<{ tempC: number; summary: string }> — current conditions.
 * @example const r = await tools.getWeather({ city: "city" });
 */
declare function getWeather(args: { city: string; units?: "c" | "f" }): Promise<{ tempC: number; summary: string }>;

// tools.echoText
/**
 * Echoes text back.
 * @param args.value — a plain string. (required)
 * @returns Promise<string> — plain text result.
 * @example const r = await tools.echoText({ value: "value" });
 */
declare function echoText(args: { value: string }): Promise<string>;

// tools.github.createIssue
/**
 * Creates a GitHub issue.
 * @param args.title — the issue title. (required)
 * @returns Promise<string> — plain text result.
 * @example const r = await tools.github.createIssue({ title: "title" });
 */
declare function createIssue(args: { title: string }): Promise<string>;

// tools.github.search
/**
 * Searches GitHub issues.
 * @param args.query — the search query. (required)
 * @returns Promise<string> — plain text result.
 * @example const r = await tools.github.search({ query: "query" });
 */
declare function search(args: { query: string }): Promise<string>;
