module counters
  implicit none
  type :: counter
    integer :: count = 0
  contains
    procedure :: increment
    procedure :: value
  end type counter
contains
  subroutine increment(self)
    class(counter), intent(inout) :: self
    self%count = self%count + 1
  end subroutine increment

  function value(self) result(res)
    class(counter), intent(in) :: self
    integer :: res
    res = self%count
  end function value
end module counters
