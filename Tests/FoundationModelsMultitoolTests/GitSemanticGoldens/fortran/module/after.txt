module geometry
  implicit none
contains
  subroutine scale(width, height, factor)
    real, intent(inout) :: width, height
    real, intent(in) :: factor
    width = width * factor
    height = height * factor
  end subroutine scale

  subroutine swap(width, height)
    real, intent(inout) :: width, height
    real :: kept
    kept = width
    width = height
    height = kept
  end subroutine swap
end module geometry
